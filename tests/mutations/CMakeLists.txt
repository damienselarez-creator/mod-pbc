cmake_minimum_required(VERSION 3.16)
project(pbc_mutation_tests LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/character/pbc_character.cpp" source)
string(FIND "${source}" "PBC_HistoryResult PBC_UpdateHistoryMessage(" first)
string(FIND "${source}" "// Snapshot var substitution helper" last)
math(EXPR length "${last} - ${first}")
string(SUBSTRING "${source}" ${first} ${length} mutations)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_mutations.inc" "${mutations}")
add_executable(mutation_test mutation_test.cpp)
target_include_directories(mutation_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(mutation_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(mutation_test PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(mutation_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME mutation_publication COMMAND mutation_test)
set_tests_properties(mutation_publication PROPERTIES TIMEOUT 30)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/api/pbc_http_handlers.cpp" http)
string(FIND "${http}" "static bool RespondMutationResult(" first)
string(FIND "${http}" "// Extract a required string field" last)
math(EXPR length "${last} - ${first}")
string(SUBSTRING "${http}" ${first} ${length} response)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_response.inc" "${response}")
