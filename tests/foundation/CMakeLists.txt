cmake_minimum_required(VERSION 3.16)
project(pbc_foundation_tests LANGUAGES CXX)
enable_testing()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
configure_file(../../src/llm/pbc_llm.cpp production_llm.cpp COPYONLY)
configure_file(../../src/core/pbc_http_client.cpp production_http.cpp COPYONLY)
add_executable(llm_test llm_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/production_llm.cpp")
add_executable(transport_test transport_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/production_http.cpp")
foreach(target llm_test transport_test)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_include_directories(${target} PRIVATE stubs ../../src/llm ../../deps ../../deps/yhirose/cpp-httplib)
 if(MSVC)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
 endif()
endforeach()
target_compile_definitions(transport_test PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(transport_test PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
 target_link_libraries(transport_test PRIVATE ws2_32 crypt32)
endif()
find_program(OPENSSL_COMMAND openssl HINTS "C:/Program Files/OpenSSL-Win64/bin" REQUIRED)
execute_process(COMMAND "${OPENSSL_COMMAND}" req -x509 -newkey rsa:2048 -nodes -days 2
 -keyout "${CMAKE_CURRENT_BINARY_DIR}/key.pem" -out "${CMAKE_CURRENT_BINARY_DIR}/cert.pem"
 -subj /CN=localhost -addext subjectAltName=DNS:localhost,IP:127.0.0.1
 RESULT_VARIABLE certificate_result OUTPUT_QUIET ERROR_QUIET)
if(NOT certificate_result EQUAL 0)
 message(FATAL_ERROR "Could not generate local test certificate")
endif()
execute_process(COMMAND "${OPENSSL_COMMAND}" req -x509 -newkey rsa:2048 -nodes -days 2
 -keyout "${CMAKE_CURRENT_BINARY_DIR}/wrong-key.pem" -out "${CMAKE_CURRENT_BINARY_DIR}/wrong-cert.pem"
 -subj /CN=wrong.invalid -addext subjectAltName=DNS:wrong.invalid
 RESULT_VARIABLE certificate_result OUTPUT_QUIET ERROR_QUIET)
if(NOT certificate_result EQUAL 0)
 message(FATAL_ERROR "Could not generate wrong-host certificate")
endif()
add_test(NAME llm_completion_contract COMMAND llm_test)
add_test(NAME transport_http COMMAND transport_test http)
add_test(NAME transport_security COMMAND transport_test "${CMAKE_CURRENT_BINARY_DIR}/cert.pem" "${CMAKE_CURRENT_BINARY_DIR}/key.pem" "${CMAKE_CURRENT_BINARY_DIR}/wrong-cert.pem" "${CMAKE_CURRENT_BINARY_DIR}/wrong-key.pem")
set_tests_properties(llm_completion_contract transport_security transport_http PROPERTIES TIMEOUT 30)
set_tests_properties(transport_security PROPERTIES SKIP_RETURN_CODE 77)

add_executable(http_library_test http_library_test.cpp)
target_compile_features(http_library_test PRIVATE cxx_std_17)
target_compile_definitions(http_library_test PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
target_include_directories(http_library_test PRIVATE ../../deps/yhirose/cpp-httplib)
target_link_libraries(http_library_test PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
    target_link_libraries(http_library_test PRIVATE ws2_32 crypt32)
endif()
add_test(NAME http_library_header_and_chunk_regressions COMMAND http_library_test http)
add_test(NAME http_library_websocket_roundtrip COMMAND http_library_test
    "${CMAKE_CURRENT_BINARY_DIR}/cert.pem" "${CMAKE_CURRENT_BINARY_DIR}/key.pem")
set_tests_properties(http_library_header_and_chunk_regressions http_library_websocket_roundtrip
    PROPERTIES TIMEOUT 15)


file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/events/pbc_event_dispatch.cpp" events)
string(FIND "${events}" "bool PBC_EventExpired(" first)
string(FIND "${events}" "// ---------------------------------------------------------------------------" last REVERSE)
string(SUBSTRING "${events}" ${first} -1 functions)
string(FIND "${functions}" "// ---------------------------------------------------------------------------" end)
string(SUBSTRING "${functions}" 0 ${end} functions)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/core/pbc_config.cpp" config)
string(FIND "${config}" "std::shared_ptr<const PBC_APIConfig> PBC_GetConnection(" first)
string(FIND "${config}" "void PBC_LoadConfig(" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${config}" ${first} ${size} connection)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_events.inc" "${functions}\n${connection}")
add_executable(event_test event_test.cpp)
target_compile_features(event_test PRIVATE cxx_std_17)
target_include_directories(event_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(event_test PRIVATE Threads::Threads)
add_test(NAME event_admission_and_connections COMMAND event_test)
set_tests_properties(event_admission_and_connections PROPERTIES TIMEOUT 15)

string(FIND "${processor}" "bool ProcessNormal(" last)
