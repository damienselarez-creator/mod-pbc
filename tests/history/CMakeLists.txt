cmake_minimum_required(VERSION 3.16)
project(pbc_history_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/character/pbc_character.cpp" source)
string(FIND "${source}" "uint64_t PBC_AppendHistoryMessage(" first)
string(FIND "${source}" "int PBC_EstimateHistoryTokens(" last)
math(EXPR length "${last} - ${first}")
string(SUBSTRING "${source}" ${first} ${length} append)
string(FIND "${source}" "bool PBC_MaybeInsertTimeGap(" first)
string(FIND "${source}" "std::unordered_set<uint64_t> PBC_MaybeInsertSharedTimeGap(" last)
math(EXPR length "${last} - ${first}")
string(SUBSTRING "${source}" ${first} ${length} gap)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_history.inc" "${append}\n${gap}")
add_executable(history_test history_test.cpp)
target_include_directories(history_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(history_test PRIVATE cxx_std_17)
target_link_libraries(history_test PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(history_test PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(history_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME history_publication COMMAND history_test)
set_tests_properties(history_publication PROPERTIES TIMEOUT 30)
