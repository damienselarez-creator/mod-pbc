cmake_minimum_required(VERSION 3.16)
project(pbc_journal_tests LANGUAGES CXX)
enable_testing()
find_package(OpenSSL REQUIRED)
add_executable(journal_test journal_test.cpp ../../src/core/pbc_history_journal.cpp)
target_include_directories(journal_test PRIVATE ../../src/core ../../src/llm ../../deps)
target_compile_features(journal_test PRIVATE cxx_std_17)
target_link_libraries(journal_test PRIVATE OpenSSL::Crypto)
if(MSVC)
    target_compile_options(journal_test PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME journal_crash_write COMMAND journal_test write "${CMAKE_CURRENT_BINARY_DIR}/journal")
set_tests_properties(journal_crash_write PROPERTIES FIXTURES_SETUP pending_journal)
add_test(NAME journal_recover COMMAND journal_test recover "${CMAKE_CURRENT_BINARY_DIR}/journal")
set_tests_properties(journal_recover PROPERTIES FIXTURES_REQUIRED pending_journal)
