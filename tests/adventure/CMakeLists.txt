cmake_minimum_required(VERSION 3.16)
project(pbc_adventure_tests LANGUAGES CXX)
enable_testing()
find_package(OpenSSL REQUIRED)
add_executable(adventure_test adventure_test.cpp
  ../../src/character/pbc_adventure_store.cpp ../../src/core/pbc_history_journal.cpp)
target_compile_features(adventure_test PRIVATE cxx_std_17)
target_include_directories(adventure_test PRIVATE ../../src/character ../../src/core ../../src/llm ../../deps)
target_link_libraries(adventure_test PRIVATE OpenSSL::Crypto)
if(MSVC)
  target_compile_options(adventure_test PRIVATE /utf-8 /W4)
endif()
add_test(NAME adventure_persistence_and_synthesis COMMAND adventure_test)
set_tests_properties(adventure_persistence_and_synthesis PROPERTIES TIMEOUT 45)
